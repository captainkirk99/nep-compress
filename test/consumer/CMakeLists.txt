cmake_minimum_required(VERSION 3.18)
project(nepcompress_consumer LANGUAGES C Fortran)
find_package(NEPCompress REQUIRED)
add_executable(consumer_c consumer.c)
target_link_libraries(consumer_c PRIVATE NEPCompress::nepcompress)
if(TARGET NEPCompress::nepcompressf)
    add_executable(consumer_fortran consumer.f90)
    target_link_libraries(consumer_fortran PRIVATE NEPCompress::nepcompressf)
endif()
